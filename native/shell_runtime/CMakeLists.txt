cmake_minimum_required(VERSION 3.20)
project(pal_shell_prototype LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT UNIX)
  message(FATAL_ERROR "The shell prototype currently targets Linux and macOS")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module Development.Embed)
find_package(Threads REQUIRED)
set(FLUX_FOUNDRY_INCLUDE_DIR "" CACHE PATH "Flux Foundry checkout (required)")
set(DYNABRIDGE_INCLUDE_DIR "" CACHE PATH "Dyna Bridge checkout, including GIL lifetime fix (required)")
foreach(path IN ITEMS "${FLUX_FOUNDRY_INCLUDE_DIR}/flow/flow.h" "${DYNABRIDGE_INCLUDE_DIR}/dynabridge/bridge.h")
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing dependency: ${path}")
  endif()
endforeach()
# Prefer an explicitly supplied static archive; never silently ship a shared libuv dependency.
set(LIBUV_STATIC_LIBRARY "" CACHE FILEPATH "PIC libuv static archive")
find_path(LIBUV_INCLUDE_DIR uv.h REQUIRED)
if(NOT LIBUV_STATIC_LIBRARY)
  set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
  find_library(PAL_UV_ARCHIVE NAMES uv libuv REQUIRED)
  set(LIBUV_STATIC_LIBRARY "${PAL_UV_ARCHIVE}")
endif()
Python3_add_library(_pal_shell_runtime MODULE WITH_SOABI
  module.cpp runtime.cpp process_posix.cpp)
target_include_directories(_pal_shell_runtime PRIVATE . "${DYNABRIDGE_INCLUDE_DIR}" "${FLUX_FOUNDRY_INCLUDE_DIR}" "${LIBUV_INCLUDE_DIR}")
target_link_libraries(_pal_shell_runtime PRIVATE "${LIBUV_STATIC_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(_pal_shell_runtime PRIVATE util rt)
  target_link_options(_pal_shell_runtime PRIVATE "LINKER:--exclude-libs,ALL")
endif()
target_compile_options(_pal_shell_runtime PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(_pal_shell_runtime PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
enable_testing()
add_test(NAME shell_native_and_host
  COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -v)
set_tests_properties(shell_native_and_host PROPERTIES
  ENVIRONMENT "PYTHONPATH=$<TARGET_FILE_DIR:_pal_shell_runtime>:${CMAKE_CURRENT_SOURCE_DIR}/python:${CMAKE_CURRENT_SOURCE_DIR}/../../src"
  TIMEOUT 120)
# Exercise the bridge fix with actual Python-owned task captures, not just shell callbacks.
if(NOT EXISTS "${DYNABRIDGE_INCLUDE_DIR}/tests/python_executor_lifetime.cpp")
  message(FATAL_ERROR "Apply dependencies/bridge-gil-lifetime.patch to the pinned bridge checkout first")
endif()
add_executable(shell_gil_lifetime "${DYNABRIDGE_INCLUDE_DIR}/tests/python_executor_lifetime.cpp")
target_include_directories(shell_gil_lifetime PRIVATE "${DYNABRIDGE_INCLUDE_DIR}" "${FLUX_FOUNDRY_INCLUDE_DIR}")
target_link_libraries(shell_gil_lifetime PRIVATE Python3::Python Threads::Threads)
add_test(NAME shell_gil_lifetime COMMAND shell_gil_lifetime)
set_tests_properties(shell_gil_lifetime PROPERTIES TIMEOUT 20)
